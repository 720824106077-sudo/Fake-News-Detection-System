# Fake-News-Detection-System
A machine learning-based Fake News Detection System using NLP to classify news articles as real or fake.

import streamlit as st
import pickle

# Load trained model
with open("model.pkl", "rb") as file:
    model = pickle.load(file)

with open("vectorizer.pkl", "rb") as file:
    vectorizer = pickle.load(file)

# Page configuration
st.set_page_config(
    page_title="Fake News Detection System",
    page_icon="📰",
    layout="centered"
)

# Custom styling
st.markdown("""
<style>
.stApp {
    background-color: #eef4fa;
}

h1 {
    color: #153e75;
    text-align: center;
}

.result {
    padding: 20px;
    border-radius: 10px;
    text-align: center;
    font-size: 24px;
    font-weight: bold;
}
</style>
""", unsafe_allow_html=True)

# Application title
st.title("📰 Fake News Detection System")

st.write(
    "Enter a news article or headline below "
    "to predict whether it is REAL or FAKE."
)

st.divider()

# User input
news = st.text_area(
    "Enter News Content:",
    height=200,
    placeholder="Type or paste your news article here..."
)

# Prediction button
if st.button("Detect News", use_container_width=True):

    if news.strip() == "":
        st.warning("Please enter some news content.")

    else:
        # Convert text into TF-IDF features
        news_vector = vectorizer.transform([news])

        # Predict
        prediction = model.predict(news_vector)[0]

        # Display result
        if prediction == "FAKE":

            st.error("🚨 FAKE NEWS DETECTED!")

            st.markdown(
                '<div class="result">❌ FAKE NEWS</div>',
                unsafe_allow_html=True
            )

        elif prediction == "REAL":

            st.success("✅ NEWS PREDICTED AS REAL")

            st.markdown(
                '<div class="result">✅ REAL NEWS</div>',
                unsafe_allow_html=True
            )

        else:
            st.warning("Unknown prediction: " + str(prediction))

        st.info(
            "Note: This prediction is generated by a "
            "machine learning model. It does not independently "
            "verify the truth of the news."
        )

st.divider()

st.subheader("About the Project")

st.write("""
This Fake News Detection System uses Natural Language
Processing (NLP) and Machine Learning.

The system uses:
- TF-IDF Vectorization
- Passive Aggressive Classifier
- Python
- Streamlit

Its purpose is to classify news content into REAL or FAKE
based on patterns learned from the training dataset.
""")

st.caption("Mini Project | Fake News Detection System")
